import React, { useState, useEffect, useRef } from 'react';
import { MoreVertical, Trash2, AlertCircle, Loader2 } from 'lucide-react';
import { OfflineReplyManager } from '../database/OfflineReplyManager';
import { deleteSnapFromSupabase } from '../../services/supabase';

export interface MobileStoryItem {
  id: string;
  mediaUrl: string;
  type?: 'image' | 'video';
  caption?: string;
  timestamp: number;
}

export interface StoriesViewerScreenProps {
  author: {
    userId: string;
    username: string;
    fullName?: string;
    avatarUrl: string;
    department?: string;
  };
  stories: MobileStoryItem[];
  onClose: () => void;
  onOpenChatWithAuthor?: (authorId: string, replyInitialText?: string) => void;
  currentUserId?: string;
  onSnapDeleted?: (snapId: string) => void;
}

export const StoriesViewerScreen: React.FC<StoriesViewerScreenProps> = ({
  author,
  stories,
  onClose,
  onOpenChatWithAuthor,
  currentUserId,
  onSnapDeleted,
}) => {
  const [currentIndex, setCurrentIndex] = useState(0);
  const [progress, setProgress] = useState(0);
  const [isPaused, setIsPaused] = useState(false);
  const [replyInput, setReplyInput] = useState('');
  const [showHeartBurst, setShowHeartBurst] = useState(false);
  const [reactionsSent, setReactionsSent] = useState<string[]>([]);
  const [isOfflineQueued, setIsOfflineQueued] = useState(false);
  const [showMenu, setShowMenu] = useState(false);
  const [showConfirmDelete, setShowConfirmDelete] = useState(false);
  const [isDeleting, setIsDeleting] = useState(false);
  const [deleteError, setDeleteError] = useState<string | null>(null);

  const handleDeleteSnap = async () => {
    if (!currentStory) return;
    setIsDeleting(true);
    setDeleteError(null);

    try {
      const res = await deleteSnapFromSupabase(currentStory.id, currentStory.mediaUrl);
      if (res.success) {
        onSnapDeleted?.(currentStory.id);
        setIsDeleting(false);
        setShowConfirmDelete(false);
        onClose();
      } else {
        setIsDeleting(false);
        setDeleteError(res.error || 'Failed to delete snap. Please try again.');
      }
    } catch (err: any) {
      setIsDeleting(false);
      setDeleteError(err?.message || 'Failed to delete snap. Please try again.');
    }
  };

  const currentStory = stories[currentIndex] || stories[0];
  const STORY_DURATION_MS = 5000;
  const timerRef = useRef<any>(null);
  const startTimeRef = useRef<number>(Date.now());
  const pausedProgressRef = useRef<number>(0);

  // Auto progression timer
  useEffect(() => {
    if (!currentStory) return;

    if (isPaused) {
      clearInterval(timerRef.current);
      return;
    }

    startTimeRef.current = Date.now() - pausedProgressRef.current * STORY_DURATION_MS;

    timerRef.current = setInterval(() => {
      const elapsed = Date.now() - startTimeRef.current;
      const newProgress = Math.min(1, elapsed / STORY_DURATION_MS);
      setProgress(newProgress);

      if (newProgress >= 1) {
        clearInterval(timerRef.current);
        handleNextStory();
      }
    }, 50);

    return () => clearInterval(timerRef.current);
  }, [currentIndex, isPaused]);

  const handleNextStory = () => {
    if (currentIndex < stories.length - 1) {
      setCurrentIndex((prev) => prev + 1);
      setProgress(0);
      pausedProgressRef.current = 0;
    } else {
      onClose();
    }
  };

  const handlePrevStory = () => {
    if (currentIndex > 0) {
      setCurrentIndex((prev) => prev - 1);
      setProgress(0);
      pausedProgressRef.current = 0;
    } else {
      setProgress(0);
      pausedProgressRef.current = 0;
    }
  };

  const handlePressIn = () => {
    pausedProgressRef.current = progress;
    setIsPaused(true);
  };

  const handlePressOut = () => {
    setIsPaused(false);
  };

  const handleSendReaction = async (emoji: string) => {
    setReactionsSent((prev) => [...prev, emoji]);
    setShowHeartBurst(true);
    setTimeout(() => setShowHeartBurst(false), 900);

    // Queue in WatermelonDB offline manager
    try {
      await OfflineReplyManager.enqueueReply(currentStory.id, author.userId, 'like', emoji);
      setIsOfflineQueued(true);
      setTimeout(() => setIsOfflineQueued(false), 2000);
    } catch (e) {
      console.warn('Queued reaction error:', e);
    }
  };

  const handleSendReply = async (e?: React.FormEvent) => {
    if (e) e.preventDefault();
    if (!replyInput.trim()) return;

    const text = replyInput.trim();
    setReplyInput('');

    try {
      await OfflineReplyManager.enqueueReply(currentStory.id, author.userId, 'reply', text);
      setIsOfflineQueued(true);
      setTimeout(() => setIsOfflineQueued(false), 2500);

      if (onOpenChatWithAuthor) {
        onOpenChatWithAuthor(author.userId, `Replying to status: "${text}"`);
      }
    } catch (e) {
      console.warn('Queue reply error:', e);
    }
  };

  return (
    <div className="fixed inset-0 z-50 bg-black flex flex-col items-center justify-center select-none overflow-hidden touch-none">
      {/* Container simulating native phone viewport */}
      <div className="relative w-full max-w-md h-full max-h-[860px] bg-neutral-950 flex flex-col justify-between overflow-hidden shadow-2xl">
        {/* Story Media Background */}
        <div
          className="absolute inset-0 z-0 bg-neutral-900"
          onMouseDown={handlePressIn}
          onMouseUp={handlePressOut}
          onTouchStart={handlePressIn}
          onTouchEnd={handlePressOut}
        >
          {currentStory?.mediaUrl ? (
            <img
              src={currentStory.mediaUrl}
              alt="Story"
              className="w-full h-full object-cover"
            />
          ) : (
            <div className="w-full h-full flex items-center justify-center bg-gradient-to-br from-indigo-900 via-neutral-900 to-amber-950 p-6 text-center text-white">
              <p className="text-xl font-medium leading-relaxed">{currentStory?.caption || 'Campus Update'}</p>
            </div>
          )}

          {/* Touch navigation zones */}
          <div
            className="absolute top-16 left-0 w-1/3 bottom-24 z-10 cursor-pointer"
            onClick={handlePrevStory}
          />
          <div
            className="absolute top-16 right-0 w-2/3 bottom-24 z-10 cursor-pointer"
            onClick={handleNextStory}
          />

          {/* Subtle gradient vignette for text legibility */}
          <div className="absolute inset-0 pointer-events-none bg-gradient-to-t from-black/80 via-transparent to-black/60" />
        </div>

        {/* Top Header & Segmented Progress Bars */}
        <div className="relative z-20 pt-4 px-3 space-y-2.5">
          {/* Segmented Progress Bars (WhatsApp / Instagram style) */}
          <div className="flex gap-1.5 w-full">
            {stories.map((st, idx) => {
              let fill = 0;
              if (idx < currentIndex) fill = 100;
              else if (idx === currentIndex) fill = progress * 100;
              return (
                <div key={st.id} className="h-1 flex-1 bg-white/30 rounded-full overflow-hidden backdrop-blur-sm">
                  <div
                    className="h-full bg-white rounded-full transition-all duration-75"
                    style={{ width: `${fill}%` }}
                  />
                </div>
              );
            })}
          </div>

          {/* User Info Bar */}
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-2.5">
              <img
                src={author.avatarUrl}
                alt={author.username}
                className="w-10 h-10 rounded-full border-2 border-amber-500 object-cover shadow-md"
              />
              <div>
                <div className="flex items-center gap-1.5">
                  <span className="text-sm font-bold text-white tracking-wide">{author.username}</span>

              {currentUserId && currentUserId === author.userId && (

                  <button

                    title="Snap Options"
                  >
                    <MoreVertical className="h-4 w-4" />

                    <div className="absolute right-0 top-10 w-44 rounded-2xl bg-neutral-900 border border-neutral-800 shadow-2xl p-1.5 z-30 animate-in fade-in duration-150">

/**
 * CampusFoundry Mobile - Biometric Security Service
 * Scaffolds Expo LocalAuthentication (FaceID, TouchID, Android Biometrics)
 * with graceful fallback for web and development simulation.
 */

export interface BiometricAuthResult {
  success: boolean;
  error?: string;
  biometricType?: 'FACE_ID' | 'FINGERPRINT' | 'IRIS' | 'NONE';
}

const STORAGE_BIOMETRIC_PREF_KEY = 'campusfoundry_biometric_enabled';
const STORAGE_BIOMETRIC_TOKEN_KEY = 'campusfoundry_biometric_token';

export class BiometricsService {
  /**
   * Checks whether the device hardware supports biometric authentication
   */
  static async isHardwareSupported(): Promise<boolean> {
    try {
      // In production Expo React Native:
      // const LocalAuthentication = require('expo-local-authentication');
      // return await LocalAuthentication.hasHardwareAsync();
      if (typeof window !== 'undefined' && window.PublicKeyCredential) {
        return true;
      }
      return true; // Supported in mobile context
    } catch {
      return false;
    }
  }

  /**
   * Checks whether biometric credentials (fingerprint/face) are enrolled
   */
  static async isEnrolled(): Promise<boolean> {
    try {
      // In production Expo React Native:
      // const LocalAuthentication = require('expo-local-authentication');
      // return await LocalAuthentication.isEnrolledAsync();
      return true;
    } catch {
      return false;
    }
  }

  /**
   * Gets the active biometric sensor type (Face ID vs Fingerprint)
   */
  static async getBiometricType(): Promise<'FACE_ID' | 'FINGERPRINT' | 'NONE'> {
    // Detect iOS vs Android user-agent or native platform
    const isIOS = typeof navigator !== 'undefined' && /iPhone|iPad|iPod/i.test(navigator.userAgent);
    return isIOS ? 'FACE_ID' : 'FINGERPRINT';
  }

  /**
   * Prompts user for biometric unlock
   */
  static async authenticateUser(promptMessage: string = 'Scan Face ID or Fingerprint to unlock CampusFoundry'): Promise<BiometricAuthResult> {
    const type = await this.getBiometricType();

    try {
      // In production Expo:
      // const LocalAuthentication = require('expo-local-authentication');
      // const result = await LocalAuthentication.authenticateAsync({
      //   promptMessage,
      //   fallbackLabel: 'Use UNILORIN Student Password / OTP',
      //   disableDeviceFallback: false,
      // });
      // return { success: result.success, biometricType: type, error: result.error };

      // In web/preview environment, simulate biometric prompt
      const isEnabled = this.isBiometricsEnabled();
      if (!isEnabled) {
        return {
          success: false,
          error: 'Biometrics not yet enabled for this device',
          biometricType: type,
        };
      }

      return {
        success: true,
        biometricType: type,
      };
    } catch (err: any) {
      return {
        success: false,
        error: err?.message || 'Biometric authentication failed',
        biometricType: type,
      };
    }
  }

  /**
   * Enables or disables biometrics for the logged-in student
   */
  static setBiometricsEnabled(enabled: boolean, studentEmail?: string): void {
    try {
      if (typeof localStorage !== 'undefined') {
        localStorage.setItem(STORAGE_BIOMETRIC_PREF_KEY, JSON.stringify(enabled));
        if (enabled && studentEmail) {
          localStorage.setItem(STORAGE_BIOMETRIC_TOKEN_KEY, studentEmail);
        } else if (!enabled) {
          localStorage.removeItem(STORAGE_BIOMETRIC_TOKEN_KEY);
        }
      }
    } catch (e) {
      console.warn('Storage error saving biometric preference:', e);
    }
  }

  /**
   * Returns whether biometrics is active for this device
   */
  static isBiometricsEnabled(): boolean {
    try {
      if (typeof localStorage !== 'undefined') {
        const val = localStorage.getItem(STORAGE_BIOMETRIC_PREF_KEY);
        return val ? JSON.parse(val) : false;
      }
    } catch {}
    return false;
  }

  /**
   * Gets the stored student email linked to biometric credentials
   */
  static getStoredBiometricEmail(): string | null {
    try {
      if (typeof localStorage !== 'undefined') {
        return localStorage.getItem(STORAGE_BIOMETRIC_TOKEN_KEY);
      }
    } catch {}
    return null;
  }
